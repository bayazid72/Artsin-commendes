<?php

namespace App\Http\Middleware;

use Closure;
use Illuminate\Http\Request;

class BeheerderMiddleware
{
    public function handle(Request $request, Closure $next)
    {
        if (! auth()->check()) {
            return redirect()->route('login');
        }

        if (auth()->user()->role !== 'beheerder') {
            abort(403);
        }

        return $next($request);
    }
}
