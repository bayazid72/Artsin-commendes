public function toewijzenForm(Melding $melding)
{
    $ids = $melding->ophalers->pluck('id');

    $vrijeOphalers = User::whereRole('ophaler')
        ->whereDoesntHave('meldingen', fn($q) => $q->where('status', 'In behandeling'))
        ->orWhereIn('id', $ids)
        ->get();

    return view('admin.toewijzen', compact('melding', 'vrijeOphalers', 'ids'));
}

public function toewijzen(Request $request, Melding $melding)
{
    $data = $request->validate([
        'ophalers' => 'required|array|min:1|max:2',
        'ophalers.*' => 'exists:users,id'
    ]);

    foreach (User::whereIn('id', $data['ophalers'])->get() as $ophaler) {
    $bezet = Melding::whereHas('ophalers', fn($q) => $q->where('users.id', $ophaler->id))
        ->where('status', 'In behandeling')
        ->where('id', '!=', $melding->id)
        ->exists();

    if ($bezet) {
        return back()->withErrors(['ophalers' => "$ophaler->name is al bezig."]);
    }
}

    $melding->ophalers()->sync($data['ophalers']);
    $melding->update(['status' => 'In behandeling']);

    return redirect()->route('admin.dashboard')->with('success', 'Ophalers bijgewerkt.');
}
